        // Historical migration evidence retains its original Data.* paths. Only this
        // test adapter maps that removed outer wrapper to the current concrete asset fields.
        static string CurrentAssetFieldPath(string historicalPath)
            => historicalPath.StartsWith("Data.", StringComparison.Ordinal) ? historicalPath.Substring(5) : historicalPath;

        public static void VerifySerialized(Object asset, Entry expected)
        {
            using var serialized = new SerializedObject(asset);
            foreach (var field in expected.Fields)
            {
                string currentPath = CurrentAssetFieldPath(field.Path);
                var property = serialized.FindProperty(currentPath);
                if (property == null)
                    throw new InvalidOperationException(expected.Id + " missing serialized field: " + field.Path + " => " + currentPath);
                bool valid;
                if (field.Kind == "Array")
                    valid = property.isArray && property.arraySize == int.Parse(field.Value, CultureInfo.InvariantCulture);
                else if (field.Kind == "Reference")
                {
                    var reference = property.objectReferenceValue;
                    valid = field.UnresolvedInBaseline ? reference == null && string.IsNullOrEmpty(AssetDatabase.GUIDToAssetPath(field.Guid)) && SavedMissingReferenceMatches(expected.Path, field) : field.FileId == 0 ? reference == null : reference != null && AssetDatabase.TryGetGUIDAndLocalFileIdentifier(reference, out string guid, out long fileId) && guid == field.Guid && fileId == field.FileId;
                }
                else if (property.propertyType == SerializedPropertyType.String)
                    valid = property.stringValue == field.Value;
                else if (property.propertyType == SerializedPropertyType.Boolean)
                    valid = property.boolValue == (field.Value == "1" || field.Value == "True");
                else if (property.propertyType == SerializedPropertyType.Float)
                {
                    double value = double.Parse(field.Value, CultureInfo.InvariantCulture);
                    valid = Math.Abs(property.doubleValue - value) <= Math.Max(0.000001, Math.Abs(value) * 0.000001);
                }
                else
                    valid = property.longValue == long.Parse(field.Value, CultureInfo.InvariantCulture);
                if (!valid)
                    throw new InvalidOperationException(expected.Id + " serialized migration mismatch: " + field.Path + " => " + currentPath);
            }
        }

        // A missing sprite already existed in the immutable baseline. Keep its exact
        // serialized PPtr while checking that import and clone retain the same missing state.
        static bool SavedMissingReferenceMatches(string assetPath, Field expected)
        {
            string[] parts = CurrentAssetFieldPath(expected.Path).Split('.');
            string[] lines = File.ReadAllLines(assetPath);
            int cursor = 0, indent = 2;
            for (int part = 0; part < parts.Length; part++, indent += 2)
            {
                string prefix = new string (' ', indent) + parts[part] + ":";
                bool found = false;
                for (; cursor < lines.Length; cursor++)
                {
                    string line = lines[cursor];
                    if (!line.StartsWith(prefix, StringComparison.Ordinal))
                        continue;
                    found = true;
                    cursor++;
                    if (part == parts.Length - 1)
                    {
                        var pointer = Regex.Match(line, @"\{fileID: (-?\d+), guid: ([0-9a-f]+), type: \d+\}");
                        return pointer.Success && pointer.Groups[2].Value == expected.Guid && long.Parse(pointer.Groups[1].Value, CultureInfo.InvariantCulture) == expected.FileId;
                    }

                    break;
                }

                if (!found)
                    return false;
            }

            return false;
        }

        static void VerifyQuestScaling(GameObject template, Action<bool, string> check)
        {
            var content = template.GetComponent<GameContentSetAuthoring>().Content;
            using var blob = QuestCatalogBaking.Compile(content);
            for (int i = 0; i < content.Quests.Definitions.Length; i++)
            {
                var source = content.Quests.Definitions[i];
                ref var runtime = ref blob.Value.Definitions[i];
                var owned = source.Objectives.OwnedItemObjectives.OrderBy(x => x.Order).ToArray();
                var submitted = source.Objectives.SubmittedItemObjectives.OrderBy(x => x.Order).ToArray();
                var rewards = source.Rewards.Items.OrderBy(x => x.Order).ToArray();
                var failures = source.FailurePenalties.OrderBy(x => x.Order).ToArray();
                for (int n = 0; n < owned.Length; n++)
                    check(runtime.Objectives.OwnedItemObjectives[n].Quantity == (int)Unity.Mathematics.math.round(owned[n].Quantity * source.ItemQuantityScale), "Quest owned-item quantity scales once");
                for (int n = 0; n < submitted.Length; n++)
                    check(runtime.Objectives.SubmittedItemObjectives[n].Quantity == (int)Unity.Mathematics.math.round(submitted[n].Quantity * source.ItemQuantityScale), "Quest submitted-item quantity scales once");
                for (int n = 0; n < rewards.Length; n++)
                    check(runtime.Rewards.Items[n].Quantity == (int)Unity.Mathematics.math.round(rewards[n].Quantity * source.ItemQuantityScale), "Quest item reward quantity scales once");
                for (int n = 0; n < failures.Length; n++)
                    check(runtime.FailurePenalties[n].Quantity == (int)Unity.Mathematics.math.round(failures[n].Quantity * source.ItemQuantityScale), "Quest item penalty quantity scales once");
            }
        }
